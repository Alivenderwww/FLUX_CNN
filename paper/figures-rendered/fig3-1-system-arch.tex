    core box/.style={
        draw=coAnno, line width=0.8pt, rounded corners=2pt,

    smc box/.style={

    },
    ddr box/.style={
        draw=coAnno, line width=0.8pt, rounded corners=2pt,
        fill=flGray, minimum width=28mm, minimum height=10mm,

    },
    bus thin/.style={
        -{Stealth[length=3mm,width=2mm]},

        line width=1.4pt, draw=coBaseline,
    },
    push dash/.style={
        -{Stealth[length=3mm,width=2mm]},

% =============================================================================
% Layer 1: Host CPU
% =============================================================================
\node[host box] (host) at (-3.5, 5.5) {Host CPU};
\node[font=\sffamily\scriptsize\itshape, text=coAnno, anchor=west]
    at ($(host.east)+(0.3,0)$) {ARM / RISC-V，启动期写 4 reg + start\_dfe};

% =============================================================================
% Layer 2: axi_lite_1to4 CSR 分发
% =============================================================================
\node[csr box] (csr) at (0, 4.0) {axi\_lite\_1to4};

\draw[bus thin] (host.south) |- ($(csr.west)+(-0.5,0)$) -- (csr.west);
\node[flow note] at (-2.0, 4.45) {AXI-Lite 32 b};

% =============================================================================
% Layer 3: 4 × core_top
% =============================================================================
\foreach \i/\x in {0/-3.6, 1/-1.2, 2/1.2, 3/3.6}{
    \node[core box] (c\i) at (\x, 1.8)
        {Core[\i]\\\scriptsize\mdseries core\_top};
    \draw[bus thin]
        ($(csr.south)+({\x*0.95},0)$) -- ($(c\i.north)+(0,0)$);
}

% Core[0] 子模块清单 (标在 Core[0] 下方)
\node[font=\sffamily\scriptsize, text=coAnno, anchor=north, align=center]
    at ($(c0.south)+(0,-0.05)$)
    {cfg\_regs / sequencer\\line\_buf / mac\_array\\parf\_accum / ofb\_writer\\wgt\_buffer\\(i/w/o/r)dma\_ctrl\\mm2s\_arb / axi\_dm\\axi\_m\_mux / ifb\_axi\_slv};
\foreach \i in {1,2,3}{
    \node[font=\sffamily\scriptsize\itshape, text=coGuide, anchor=north]
        at ($(c\i.south)+(0,-0.05)$)
        {(同 Core[0])};
}

% =============================================================================
% Layer 4: SmartConnect 4×4
% =============================================================================
\node[smc box] (smc) at (0, -1.6) {SmartConnect \ 4-Master $\times$ 4-Slave};

\foreach \i/\x in {0/-3.6, 1/-1.2, 2/1.2, 3/3.6}{
    \draw[bus thick]
        ($(c\i.south)+(0,-2.1)$) -- ($(smc.north)+({\x*0.95},0)$);
}
\node[flow note] at (4.5, 0.5) {AXI4 \ 128 b};

% =============================================================================
% 跨核 SRAM push (红虚): SmartConnect 反向写入 core.ifb_axi_slv
% =============================================================================
\foreach \i/\x in {0/-3.6, 1/-1.2, 2/1.2, 3/3.6}{
    \draw[push dash]
        ($(smc.north)+({\x*0.95+0.3},0)$) to[bend right=12]
        ($(c\i.east)+(0,-0.1)$);
}
\node[font=\sffamily\scriptsize, text=coAccent, anchor=west, align=left]
    at (4.8, -0.6)
    {跨核 SRAM\\push (红虚)};

% =============================================================================
% Layer 5: Off-chip DDR
% =============================================================================
\node[ddr box] (ddr) at (0, -4.0) {Off-chip DDR};

\draw[bus thick] (smc.south) -- (ddr.north);
\node[flow note] at (1.6, -2.8) {AXI4 \ 128 b};

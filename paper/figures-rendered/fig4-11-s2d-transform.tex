\foreach \r in {0,1,2,3} {
  \foreach \c in {0,1,2,3} {
    \pgfmathtruncatemacro{\rp}{int(mod(\r,2))}
    \pgfmathtruncatemacro{\cp}{int(mod(\c,2))}
    \ifnum\rp=0
      \ifnum\cp=0
        \def\sty{px tl}
      \else
        \def\sty{px tr}
      \fi
    \else
      \ifnum\cp=0
        \def\sty{px bl}
      \else
        \def\sty{px br}
      \fi
    \fi
    \node[\sty] at ({\cx-1.05+\c*0.7},{11.2-\r*0.7}) {};
  }
}

\begin{document}
\begin{tikzpicture}[
    every node/.style={font=\sffamily\footnotesize},
    pe slot/.style={
        draw=coAnno, line width=0.4pt, rectangle,
        minimum width=6mm, minimum height=6mm,
        inner sep=0pt, font=\sffamily\scriptsize,
        align=center,
    },
    pe used/.style={pe slot, fill=flBlue},
    pe idle/.style={pe slot, fill=white, text=coGuide},
    pe g1/.style={pe slot, fill=flBlue},
    pe g2/.style={pe slot, fill=flYellow},
    pe g3/.style={pe slot, fill=flGreen},
    pe g4/.style={pe slot, fill=flGray},
    pe g5/.style={pe slot, fill=coBaseline!18},
    row title/.style={
        font=\sffamily\small\bfseries, text=coAnno,
        anchor=east,
    },
    util label/.style={
        font=\sffamily\footnotesize\bfseries,
        anchor=north,
    },
    fold arrow/.style={
        -{Stealth[length=4mm,width=2.8mm]},
        line width=1.8pt,
        draw=coAccent,
    },
    formula box/.style={
        draw=coAnno, line width=0.4pt, fill=white, rounded corners=1pt,

        font=\sffamily\footnotesize,
    },

% =============================================================================
% 横排 16 PE: 每 PE 6mm 宽, 起点 x0, y 由 row 决定
% 用 \pgfmathsetmacro 算坐标
% =============================================================================
\def\xz{0.0}        % 起点 x
\def\dx{0.7}        % PE 间距 (cm)
\def\yTop{4.0}      % 折叠前行 y
\def\yBot{0.0}      % 折叠后行 y

% ---- 折叠前: PE 0..2 used (blue), PE 3..15 idle ----
\node[row title] at (\xz-0.4, \yTop)
    {$\mathbf{C_{in}{=}3}$};

\foreach \i in {0,1,2}{\node[pe used] at ({\xz+\i*\dx}, \yTop) {\i};}
\foreach \i in {3,...,15}{\node[pe idle] at ({\xz+\i*\dx}, \yTop) {/};}

\node[util label, text=coAnno] at ({\xz}, \yTop-0.45)
    {Util = 3 / 16 = 18.75\%};

% ---- 折叠后: 5 组 × 3 (ky=0..4, ci=0..2) + 1 idle ----
\node[row title] at (\xz-0.4, \yBot)
    {$\mathbf{C_{in}'{=}15}$};

\foreach \i/\sty in {%
    0/pe g1, 1/pe g1, 2/pe g1,
    3/pe g2, 4/pe g2, 5/pe g2,
    6/pe g3, 7/pe g3, 8/pe g3,
    9/pe g4, 10/pe g4, 11/pe g4,
    12/pe g5, 13/pe g5, 14/pe g5}{
    \node[\sty] at ({\xz+\i*\dx}, \yBot) {\i};
}
\node[pe idle] at ({\xz+15*\dx}, \yBot) {15};

\node[util label, text=coImproved] at ({\xz}, \yBot+0.9)
    {Util = 15 / 16 = 93.75\%};

% ---- 5 组颜色图例 (在折叠后行下方) ----
\node[font=\sffamily\scriptsize, text=coAnno, anchor=north]
    at ({\xz+7*\dx}, \yBot-0.6)
    {每 3 个 PE 一组 = 一个 $k_y$，组内列 = $c_i \in \{0,1,2\}$};

% ---- 中央垂直折叠箭头 + 公式 ----
\def\yMidArrow{2.0}

\draw[fold arrow]
    ({\xz+7.5*\dx}, \yTop-0.5)
    -- ({\xz+7.5*\dx}, \yBot+0.5);

\node[formula box, anchor=west]
    at ({\xz+8.5*\dx}, \yMidArrow)
    {Ky-fold\\[2pt]
     $g_y = \lfloor 16 / C_{in} \rfloor = 5$ \\[1pt]
     $\mathrm{cin\_fake} = g_y \cdot C_{in} = 15$ \\[1pt]
     $K_{y,\mathrm{eff}} = \lceil K / g_y \rceil = 1$};

% ---- 左侧公式标签 (与右侧公式对应) ----
\node[font=\sffamily\scriptsize\bfseries, text=coAccent, anchor=east]
    at ({\xz+7*\dx}, \yMidArrow)
    {$\mathbf{K_{y}}$ into $\mathbf{C_{in}}$};

\end{tikzpicture}
